cmake_minimum_required(VERSION 3.16)
project(onestring_eigen_sparse_direct LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED EIGEN_INCLUDE_DIR OR NOT EXISTS "${EIGEN_INCLUDE_DIR}/Eigen/SparseLU")
  message(FATAL_ERROR "Set EIGEN_INCLUDE_DIR to an independently licensed Eigen distribution")
endif()
add_library(zhang2021_eigen_solver SHARED pardiso_compatible_eigen.cpp)
target_include_directories(zhang2021_eigen_solver PRIVATE "${EIGEN_INCLUDE_DIR}")
set_target_properties(zhang2021_eigen_solver PROPERTIES OUTPUT_NAME "libpardiso600-WIN-X86-64" PREFIX "")
if(MSVC)
  target_compile_options(zhang2021_eigen_solver PRIVATE /O2 /EHsc /W3)
endif()
